```python
from dataclasses import dataclass
from typing import Dict, Optional, List

@dataclass
class ScoringSummary:
    overall_score: float
    label: str
    per_feature_scores: Dict[str, float]
    missing_features: List[str]
    weighting_strategy: str

class WeightedScorer:
    def __init__(self):
        self.weights = {
            "aimbot": 0.30,
            "wallhack": 0.25,
            "triggerbot": 0.20,
            "recoil": 0.15,
            "bhop": 0.05,
            "session": 0.05,
        }
        total = sum(self.weights.values())
        assert abs(total - 1.0) < 0.001, f"Weights sum to {total}, not 1.0"
    
    def score(self, feature_results: Dict[str, FeatureResult]) -> ScoringSummary:
        # Extract valid scores
        available_scores = {}
        missing_features = []
        for feature_name, result in feature_results.items():
            if result is None or result.score is None:
                missing_features.append(feature_name)
            else:
                available_scores[feature_name] = result.score
        
        # If no features, raise error
        if not available_scores:
            raise ValueError("All features failed; cannot compute score")
        
        # Compute weighted score
        overall = sum(available_scores.get(f, 0) * self.weights.get(f, 0) 
                     for f in self.weights.keys())
        
        # Threshold mapping
        if overall < 0.3:
            label = "clean"
        elif overall < 0.7:
            label = "suspicious"
        else:
            label = "likely_cheating"
        
        return ScoringSummary(
            overall_score=overall,
            label=label,
            per_feature_scores=available_scores,
            missing_features=missing_features,
            weighting_strategy="proportional_redistribution",
        )
```

```python
def process_job(demo_id: str, file_path: str) -> None:
    # Parse
    try:
        parsed_demo = parser_adapter.parse_demo(file_path)
    except DemoParseError as e:
        log("parser_error", demo_id=demo_id, error=str(e))
        result_writer.write_error(demo_id, str(e))
        return
    
    # Extract features
    feature_results = {}
    for extractor in extractors:
        feature_name = extractor.__class__.__name__
        try:
            result = extractor.extract(parsed_demo)
            feature_results[feature_name] = result
            log("feature_extracted", demo_id=demo_id, feature=feature_name, score=result.score)
        except Exception as e:
            log("feature_error", demo_id=demo_id, feature=feature_name, error=str(e), level="warning")
            feature_results[feature_name] = None
    
    valid_results = {k: v for k, v in feature_results.items() if v is not None}
    if not valid_results:
        log("all_features_failed", demo_id=demo_id)
        result_writer.write_error(demo_id, "All features failed")
        return
    
    # Score
    try:
        scoring_summary = scorer.score(valid_results)
        log("scoring_complete", demo_id=demo_id, overall_score=scoring_summary.overall_score, label=scoring_summary.label)
    except Exception as e:
        log("scoring_error", demo_id=demo_id, error=str(e))
        result_writer.write_error(demo_id, str(e))
        return
    
    # Persist
    try:
        result_writer.write_result(demo_id, feature_results, scoring_summary)
        log("result_persisted", demo_id=demo_id)
    except Exception as e:
        log("persistence_error", demo_id=demo_id, error=str(e))
        return 1  # Exit worker
```

```python
def write_result(self, demo_id: str, feature_results: Dict[str, Optional[FeatureResult]], scoring_summary) -> None:
    # Serialize features
    featureData = {}
    for feature_name, result in feature_results.items():
        if result is None:
            featureData[feature_name] = {"error": "feature_extraction_failed", "score": None}
        else:
            featureData[feature_name] = {
                "score": result.score,
                "raw_measurements": result.raw_measurements,
                "metadata": result.metadata,
            }
    
    featureData_json = json.dumps(featureData)
    
    # Extract scores (handle None gracefully)
    scores = {f: (feature_results[f].score if feature_results[f] else None) for f in feature_results.keys()}
    
    # Build INSERT
    cursor = self.db_conn.cursor()
    try:
        cursor.execute("""
            INSERT INTO AnalysisResult (demo_id, aimbotScore, wallhackScore, triggerBotScore, recoilScore, bhopScore, sessionScore, overallSuspicion, suspicionLabel, featureData)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
        """, (
            demo_id,
            scores.get("AimbotExtractor"),
            scores.get("WallhackExtractor"),
            scores.get("TriggerbotExtractor"),
            scores.get("RecoilExtractor"),
            scores.get("BhopExtractor"),
            scores.get("SessionConsistencyExtractor"),
            scoring_summary.overall_score,
            scoring_summary.label,
            featureData_json,
        ))
        
        # Update Demo status
        cursor.execute("""
            UPDATE Demo SET status = %s WHERE id = %s
        """, ("done", demo_id))
        
        self.db_conn.commit()
        log("result_writer_success", demo_id=demo_id)
    except Exception as e:
        self.db_conn.rollback()
        log("result_writer_error", demo_id=demo_id, error=str(e))
        raise
    finally:
        cursor.close()
```
