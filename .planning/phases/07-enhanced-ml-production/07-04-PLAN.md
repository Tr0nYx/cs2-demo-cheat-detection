       def test_retry_decorator_succeeds_on_retry(self):
           """Verify retry decorator succeeds on second attempt."""
           from python.ml.model import retry_with_backoff

           attempt_count = 0

           @retry_with_backoff(max_retries=3, backoff_factor=0.01)
           def fails_once():
               nonlocal attempt_count
               attempt_count += 1
               if attempt_count < 2:
                   raise RuntimeError("First attempt fails")
               return "success"

           result = fails_once()
           assert result == "success"
           assert attempt_count == 2

       def test_retry_backoff_timing(self):
           """Verify exponential backoff timing: 1s, 2s, 4s."""
           from python.ml.model import retry_with_backoff
           import time

           @retry_with_backoff(max_retries=3, backoff_factor=0.05)
           def backoff_test():
               if not hasattr(backoff_test, 'attempts'):
                   backoff_test.attempts = 0
               backoff_test.attempts += 1
               if backoff_test.attempts < 3:
                   raise RuntimeError("Retry")
               return "success"

           start = time.time()
           result = backoff_test()
           elapsed = time.time() - start

           # Should wait ~0.05s + 0.1s = 0.15s (with some tolerance)
           assert elapsed >= 0.1 and elapsed < 0.5
           assert result == "success"
