public function findSharedByPlayers(
    string $playerAId,
    string $playerBId,
    int $limit = 10
): array {
    // Find demos where both players have analysis results
    return $this->createQueryBuilder('arA')
        ->select('DISTINCT d')
        ->from('App\Domain\Demo\Demo', 'd')
        ->innerJoin('d.analysisResults', 'arA')
        ->innerJoin('d.analysisResults', 'arB')
        ->where('arA.player = :playerA')
        ->andWhere('arB.player = :playerB')
        ->setParameter('playerA', $playerAId)
        ->setParameter('playerB', $playerBId)
        ->orderBy('d.uploadedAt', 'DESC')
        ->setMaxResults($limit)
        ->getQuery()
        ->getResult();
}

```typescript
'use client'

import { useQuery, UseQueryResult } from '@tanstack/react-query'

export interface PlayerComparisonData {
  playerAId: string
  playerAName: string
  playerAComponents: { components: Record<string, { value: number; percentile: number }>; traceDatetime: string }
  playerATrend: { history: Array<{ date: string; value: number }>; trending: boolean }
  playerAMaps: { topMaps: Array<{ map: string; traceAdjusted: number }> }
  playerAHistory: { sharedDemos: Array<{ demoId: string; date: string; mapId?: string }> }
  playerBId: string
  playerBName: string
  playerBComponents: { components: Record<string, { value: number; percentile: number }>; traceDatetime: string }
  playerBTrend: { history: Array<{ date: string; value: number }>; trending: boolean }
  playerBMaps: { topMaps: Array<{ map: string; traceAdjusted: number }> }
  playerBHistory: { sharedDemos: Array<{ demoId: string; date: string; mapId?: string }> }
}

const API_BASE_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost/api'

export function usePlayerComparison(
  playerId: string,
  compareWithId: string | null
): UseQueryResult<PlayerComparisonData | null, Error> {
  return useQuery({
    queryKey: ['playerComparison', playerId, compareWithId],
    queryFn: async (): Promise<PlayerComparisonData | null> => {
      if (!compareWithId) {
        return null
      }

      const response = await fetch(
        `${API_BASE_URL}/players/${playerId}/compare?with=${compareWithId}`
      )

      if (response.status === 404) {
        return null
      }

      if (!response.ok) {
        const errorData = await response.json().catch(() => ({}))
        throw new Error(
          errorData.message || `Failed to fetch player comparison: ${response.statusText}`
        )
      }

      return response.json() as Promise<PlayerComparisonData>
    },
    enabled: !!compareWithId, // Only run query if compareWithId is provided
    retry: 1,
    staleTime: 5 * 60 * 1000, // 5 minutes (leaderboards change as demos complete)
  })
}
```

Must:
- Use React Query useQuery hook
- Define interfaces for response data
- Handle 404 gracefully (null instead of error)
- Support enabled condition (only fetch if compareWithId present)
- Include proper error messages
- Set staleTime and retry options
- Use proper TypeScript types
