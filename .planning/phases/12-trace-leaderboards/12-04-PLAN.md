```php
class TeamRepository extends ServiceEntityRepository
{
    public function __construct(ManagerRegistry $registry)
    {
        parent::__construct($registry, Team::class);
    }

    /**
     * Find teams with qualifying members (5+ demos per member), sorted by aggregated TRACE.
     * Team score = average 95th percentile of all qualified members.
     *
     * @param int $limit Results per page
     * @param int $offset Skip first N results
     * @return array Array of Team entities with aggregated scores
     */
    public function findQualifiedTeamsAndSorted(int $limit = 100, int $offset = 0): array
    {
        // DQL approach: Join Team->Players->TraceRating, group by team, aggregate scores
        return $this->createQueryBuilder('t')
            ->select('t, AVG(tr.traceAdjusted) as HIDDEN team_score')
            ->innerJoin('t.players', 'p')
            ->leftJoin('tr.playerId = p.id', 'tr', 'WITH', 'tr.playerId = p.id')
            ->where('(SELECT COUNT(tr2.id) FROM TraceRating tr2 WHERE tr2.playerId = p.id) >= 5')
            ->groupBy('t.id')
            ->orderBy('team_score', 'DESC')
            ->setFirstResult($offset)
            ->setMaxResults($limit)
            ->getQuery()
            ->getResult();
    }

    /**
     * Count qualifying teams (with 5+ demo members).
     */
    public function countQualifiedTeams(): int
    {
        return (int) $this->createQueryBuilder('t')
            ->select('COUNT(DISTINCT t.id)')
            ->innerJoin('t.players', 'p')
            ->where('(SELECT COUNT(tr2.id) FROM TraceRating tr2 WHERE tr2.playerId = p.id) >= 5')
            ->getQuery()
            ->getSingleScalarResult();
    }

    /**
     * Get aggregated TRACE score for a team (average of member 95th percentiles).
     */
    public function getTeamAggregatedScore(string $teamId): ?float
    {
        $result = $this->createQueryBuilder('t')
            ->select('AVG(tr.traceAdjusted)')
            ->innerJoin('t.players', 'p')
            ->leftJoin('tr.playerId = p.id', 'tr')
            ->where('t.id = :teamId')
            ->andWhere('(SELECT COUNT(tr2.id) FROM TraceRating tr2 WHERE tr2.playerId = p.id) >= 5')
            ->setParameter('teamId', $teamId)
            ->getQuery()
            ->getSingleScalarResult();

        return $result ? (float) $result : null;
    }
}
```
